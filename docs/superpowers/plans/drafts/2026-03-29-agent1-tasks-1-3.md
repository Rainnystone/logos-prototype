```typescript
import { describe, expect, it } from 'vitest';
import { generateCharacterId } from '@/lib/character-id';

describe('generateCharacterId', () => {
  it('returns a string starting with chr_ prefix', () => {
    const id = generateCharacterId();
    expect(id.startsWith('chr_')).toBe(true);
  });

  it('returns exactly 10 characters total (chr_ + 6 hex)', () => {
    const id = generateCharacterId();
    expect(id).toHaveLength(10);
  });

  it('has a valid hex suffix after the prefix', () => {
    const id = generateCharacterId();
    const suffix = id.slice(4);
    expect(suffix).toMatch(/^[0-9a-f]{6}$/);
  });

  it('generates unique ids across multiple calls', () => {
    const ids = new Set(Array.from({ length: 50 }, () => generateCharacterId()));
    expect(ids.size).toBe(50);
  });

  it('matches the format chr_XXXXXX where X is lowercase hex', () => {
    const id = generateCharacterId();
    expect(id).toMatch(/^chr_[0-9a-f]{6}$/);
  });
});
```
