import { describe, expect, it } from 'vitest';
import { isLegacyWorldBase, migrateLegacyWorldBase } from '@/engine/worldbase-migration';
import { WorldBaseSchema } from '@/types/prompt-object';

const LEGACY_DATA: Record<string, unknown> = {
  mainCharacters: `## World Base Setting

不吉波普不笑世界观...

## World Rules / Prohibitions / Anomalous Properties

严格禁止一切...

## Genre Tone & Prose Baseline

日式轻小说本格风格

### Character 1

### Character 1

社会性死亡。一旦其直播间在线人数归零，所有力量瞬间溃散。`,
  npcCharacters: '竹田启司：稳重的男友',
  locationPatch: '# Location Pool',
};

describe('isLegacyWorldBase', () => {
  it('returns true when data has mainCharacters string and no hero field', () => {
    expect(isLegacyWorldBase(LEGACY_DATA)).toBe(true);
  });

  it('returns false when data has a hero field', () => {
    expect(isLegacyWorldBase({ hero: {}, locationPatch: '' })).toBe(false);
  });

  it('returns false for null/undefined/non-object', () => {
    expect(isLegacyWorldBase(null)).toBe(false);
    expect(isLegacyWorldBase(undefined)).toBe(false);
    expect(isLegacyWorldBase('string')).toBe(false);
  });
});

describe('migrateLegacyWorldBase', () => {
  it('returns a value that passes WorldBaseSchema validation', () => {
    const result = migrateLegacyWorldBase(LEGACY_DATA);
    expect(() => WorldBaseSchema.parse(result)).not.toThrow();
  });

  it('extracts world-level fields', () => {
    const result = migrateLegacyWorldBase(LEGACY_DATA);
    expect(result.worldBaseSetting).toContain('不吉波普不笑世界观');
    expect(result.worldRules).toContain('严格禁止');
    expect(result.toneBaseline).toContain('日式轻小说');
  });

  it('extracts the hero with CharacterProfile fields', () => {
    const result = migrateLegacyWorldBase(LEGACY_DATA);
    expect(result.hero.name).toBe('雾间凪');
    expect(result.hero.characterId).toMatch(/^chr_/);
  });

  it('extracts core cast and antagonists', () => {
    const result = migrateLegacyWorldBase(LEGACY_DATA);

  it('generates unique characterIds', () => {
    const result = migrateLegacyWorldBase(LEGACY_DATA);
    const allIds = [
      result.hero.characterId,
      ...result.coreCast.map((c) => c.characterId),
      ...result.antagonists.map((c) => c.characterId),
    ];
    expect(new Set(allIds).size).toBe(allIds.length);
  });

  it('preserves npcCharacters and locationPatch', () => {
    const result = migrateLegacyWorldBase(LEGACY_DATA);
    expect(result.npcCharacters).toBe('竹田启司：稳重的男友');
    expect(result.locationPatch).toBe('# Location Pool');
  });
});

### Task 5: Update story-loader with legacy detection and cast filtering

- [ ] **5.1 Write tests for `filterWorldBaseByCast`**

```typescript
import { filterWorldBaseByCast } from '@/engine/story-loader';
import type { WorldBase } from '@/types/prompt-object';

describe('filterWorldBaseByCast', () => {
  const baseWorldBase: WorldBase = {
    worldBaseSetting: 'setting',
    worldRules: 'rules',
    toneBaseline: 'tone',
    hero: { characterId: 'chr_000001', name: 'Hero', ... },
    coreCast: [
      { characterId: 'chr_000002', name: 'Ally A', ... },
      { characterId: 'chr_000003', name: 'Ally B', ... },
    ],
    antagonists: [
      { characterId: 'chr_000004', name: 'Villain', ... },
    ],
    npcCharacters: 'NPC list',
    locationPatch: 'locations',
  };

  it('returns worldBase unmodified when cast is undefined', () => {
    const result = filterWorldBaseByCast(baseWorldBase, undefined);
    expect(result).toEqual(baseWorldBase);
  });

  it('always includes the hero regardless of cast array', () => {
    const result = filterWorldBaseByCast(baseWorldBase, ['chr_000002']);
    expect(result.hero).toEqual(baseWorldBase.hero);
  });

  it('filters coreCast to only include characters in cast array', () => {
    const result = filterWorldBaseByCast(baseWorldBase, ['chr_000001', 'chr_000002']);
    expect(result.coreCast).toHaveLength(1);
    expect(result.coreCast[0]!.characterId).toBe('chr_000002');
  });

  it('filters antagonists to only include characters in cast array', () => {
    const result = filterWorldBaseByCast(baseWorldBase, ['chr_000001', 'chr_000004']);
    expect(result.antagonists).toHaveLength(1);
  });

  it('does not mutate the input worldBase', () => {
    const originalLength = baseWorldBase.coreCast.length;
    filterWorldBaseByCast(baseWorldBase, ['chr_000001']);
    expect(baseWorldBase.coreCast).toHaveLength(originalLength);
  });
});
```
